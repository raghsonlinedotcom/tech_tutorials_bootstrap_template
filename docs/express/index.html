1. Introduction

What is Express.js?

Express is a minimal and flexible Node.js web application framework providing a robust set of features for web and mobile applications.

Key Features
  • Robust routing with route parameters
  • Middleware architecture
  • Template engine support
  • HTTP utility methods
  • Content negotiation
  • High performance
Why Express?
  • Minimal and unopinionated
  • Huge middleware ecosystem
  • Industry standard (used by Netflix, Uber, PayPal)
  • Great for REST APIs
  • Active community
History

Created by TJ Holowaychuk in 2010, Express became the de facto Node.js web framework. Now maintained by the Node.js Foundation.

4. Routing

HTTP Methods
// GET - Retrieve data
app.get('/users', (req, res) => {
  res.json({ users: [] });
});

// POST - Create new resource
app.post('/users', (req, res) => {
  res.status(201).json({ message: 'User created' });
});

// PUT - Update entire resource
app.put('/users/:id', (req, res) => {
  res.json({ message: `User ${req.params.id} updated` });
});

// PATCH - Partial update
app.patch('/users/:id', (req, res) => {
  res.json({ message: `User ${req.params.id} partially updated` });
});

// DELETE - Remove resource
app.delete('/users/:id', (req, res) => {
  res.json({ message: `User ${req.params.id} deleted` });
});
Route Parameters
// Single parameter
app.get('/users/:userId', (req, res) => {
  const { userId } = req.params;
  res.json({ userId });
});

// Multiple parameters
app.get('/users/:userId/posts/:postId', (req, res) => {
  const { userId, postId } = req.params;
  res.json({ userId, postId });
});

// Optional parameters with regex
app.get('/flights/:from-:to', (req, res) => {
  res.json({ from: req.params.from, to: req.params.to });
});
Query Parameters
// URL: /search?q=express&page=1&limit=10
app.get('/search', (req, res) => {
  const { q, page = 1, limit = 10 } = req.query;
  res.json({ query: q, page, limit });
});
Route Chaining
app.route('/book')
  .get((req, res) => {
    res.send('Get books');
  })
  .post((req, res) => {
    res.send('Add a book');
  })
  .put((req, res) => {
    res.send('Update book');
  });

5. Middleware

What is Middleware?

Functions with access to request, response, and next() function in the request-response cycle.

Application-Level Middleware
// Logger middleware
app.use((req, res, next) => {
  console.log(`${new Date().toISOString()} - ${req.method} ${req.url}`);
  next(); // Pass control to next middleware
});

// Specific path middleware
app.use('/api', (req, res, next) => {
  console.log('API route accessed');
  next();
});
Built-in Middleware
// Parse JSON bodies
app.use(express.json());

// Parse URL-encoded bodies (form data)
app.use(express.urlencoded({ extended: true }));

// Serve static files
app.use(express.static('public'));
app.use('/uploads', express.static('uploads'));
Third-Party Middleware
const morgan = require('morgan');
const cors = require('cors');
const helmet = require('helmet');
const compression = require('compression');

// HTTP request logger
app.use(morgan('dev'));

// Enable CORS
app.use(cors());

// Security headers
app.use(helmet());

// Compress responses
app.use(compression());
Custom Middleware
// Authentication middleware
const requireAuth = (req, res, next) => {
  const token = req.headers.authorization;
  
  if (!token) {
    return res.status(401).json({ error: 'No token provided' });
  }
  
  // Verify token (simplified)
  try {
    // const decoded = jwt.verify(token, process.env.JWT_SECRET);
    // req.user = decoded;
    next();
  } catch (err) {
    res.status(403).json({ error: 'Invalid token' });
  }
};

// Request timing middleware
const requestTime = (req, res, next) => {
  req.requestTime = Date.now();
  next();
};

// Usage
app.use(requestTime);
app.get('/protected', requireAuth, (req, res) => {
  res.json({ message: 'Protected data', requestTime: req.requestTime });
});
Error-Handling Middleware
// Must have 4 parameters!
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(err.status || 500).json({
    error: {
      message: err.message,
      ...(process.env.NODE_ENV === 'development' && { stack: err.stack })
    }
  });
});

6. Request & Response Objects

Request Object (req)
app.get('/demo', (req, res) => {
  // URL parameters
  const id = req.params.id;
  
  // Query strings
  const search = req.query.q;
  
  // Request body (with body-parser)
  const data = req.body;
  
  // Headers
  const userAgent = req.get('User-Agent');
  const auth = req.headers.authorization;
  
  // Cookies (requires cookie-parser)
  const sessionId = req.cookies.sessionId;
  
  // Route path
  const path = req.path;        // '/demo'
  const hostname = req.hostname; // 'localhost'
  const protocol = req.protocol; // 'http'
  
  // HTTP method
  const method = req.method;     // 'GET'
  
  // IP address
  const ip = req.ip;
  
  res.json({ id, search, path, method });
});
Response Methods
// Send different response types
app.get('/api/examples', (req, res) => {
  // JSON response
  res.json({ status: 'success', data: [] });
  
  // Send string
  res.send('Hello');
  
  // Set status code
  res.status(404).send('Not found');
  
  // Redirect
  res.redirect('/new-url');
  res.redirect(301, '/moved-permanently');
  
  // Download file
  res.download('/path/to/file.pdf');
  
  // Send file
  res.sendFile(__dirname + '/index.html');
  
  // Set headers
  res.set('Content-Type', 'application/json');
  res.set({
    'Content-Type': 'text/plain',
    'X-Custom-Header': 'value'
  });
  
  // Render template
  res.render('index', { title: 'Home' });
  
  // End response
  res.end();
});

7. Error Handling

Basic Error Handling
// Synchronous errors
app.get('/sync-error', (req, res) => {
  throw new Error('Something went wrong!'); // Caught by Express
});

// Asynchronous errors (use try-catch or pass to next)
app.get('/async-error', async (req, res, next) => {
  try {
    const data = await someAsyncOperation();
    res.json(data);
  } catch (error) {
    next(error); // Pass to error handler
  }
});
Custom Error Class
class AppError extends Error {
  constructor(message, statusCode) {
    super(message);
    this.statusCode = statusCode;
    this.isOperational = true;
    Error.captureStackTrace(this, this.constructor);
  }
}

// Usage
app.get('/user/:id', async (req, res, next) => {
  const user = await User.findById(req.params.id);
  if (!user) {
    return next(new AppError('User not found', 404));
  }
  res.json(user);
});
Global Error Handler
// 404 handler
app.use((req, res, next) => {
  res.status(404).json({
    error: 'Route not found',
    path: req.originalUrl
  });
});

// Global error handler (must be last!)
app.use((err, req, res, next) => {
  const statusCode = err.statusCode || 500;
  const message = err.message || 'Internal Server Error';
  
  console.error(`[ERROR] ${statusCode} - ${message}`);
  console.error(err.stack);
  
  res.status(statusCode).json({
    error: {
      message,
      ...(process.env.NODE_ENV === 'development' && {
        stack: err.stack,
        details: err
      })
    }
  });
});
Async Error Wrapper
// Utility to catch async errors
const asyncHandler = (fn) => (req, res, next) => {
  Promise.resolve(fn(req, res, next)).catch(next);
};

// Usage
app.get('/users', asyncHandler(async (req, res) => {
  const users = await User.find();
  res.json(users); // No try-catch needed!
}));

Creating Modular Routes

models/Todo.js:

// In-memory storage (use DB in production)
let todos = [
  { id: 1, title: 'Learn Express', completed: false },
  { id: 2, title: 'Build REST API', completed: false }
];
let nextId = 3;

class Todo {
  static findAll() {
    return todos;
  }

  static findById(id) {
    return todos.find(t => t.id === parseInt(id));
  }

  static create(data) {
    const todo = { id: nextId++, ...data, completed: false };
    todos.push(todo);
    return todo;
  }

  static update(id, data) {
    const index = todos.findIndex(t => t.id === parseInt(id));
    if (index === -1) return null;
    todos[index] = { ...todos[index], ...data };
    return todos[index];
  }

  static delete(id) {
    const index = todos.findIndex(t => t.id === parseInt(id));
    if (index === -1) return false;
    todos.splice(index, 1);
    return true;
  }
}

module.exports = Todo;

controllers/todoController.js:

const Todo = require('../models/Todo');

exports.getAllTodos = (req, res) => {
  const todos = Todo.findAll();
  res.json({ count: todos.length, data: todos });
};

exports.getTodoById = (req, res, next) => {
  const todo = Todo.findById(req.params.id);
  if (!todo) {
    return res.status(404).json({ error: 'Todo not found' });
  }
  res.json({ data: todo });
};

exports.createTodo = (req, res) => {
  const { title } = req.body;
  
  if (!title || title.trim() === '') {
    return res.status(400).json({ error: 'Title is required' });
  }
  
  const todo = Todo.create({ title: title.trim() });
  res.status(201).json({ message: 'Todo created', data: todo });
};

exports.updateTodo = (req, res) => {
  const { title, completed } = req.body;
  const todo = Todo.update(req.params.id, { title, completed });
  
  if (!todo) {
    return res.status(404).json({ error: 'Todo not found' });
  }
  
  res.json({ message: 'Todo updated', data: todo });
};

exports.deleteTodo = (req, res) => {
  const deleted = Todo.delete(req.params.id);
  
  if (!deleted) {
    return res.status(404).json({ error: 'Todo not found' });
  }
  
  res.json({ message: 'Todo deleted' });
};
